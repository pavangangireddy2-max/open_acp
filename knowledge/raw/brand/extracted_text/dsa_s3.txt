                           Space Complexity



                   Measures the memory an algorithm

Example - Constant Space

                             Code

   #include <iostream>
   using namespace std;


   int main() {
       int a = 10;                       Constant space
       int b = 20;
       int sum = a + b;
                                           O(1)
       cout << "Sum: " << sum << endl;
       return 0;
   }

Example - Linear Space

                                       Code

   #include <bits/stdc++.h>
   using namespace std;


   int main() {                                The program reads an integer n from
       int n;
       cin >> n;                              the user and creates a vector of size n, so
       vector<int> arr(n);
       for (int i = 0; i < n; i++) {
                                               the space used grows linearly with n.
           arr[i] = i;
       }
       for (int i = 0; i < n; i++) {                            O(n)
           cout << arr[i] << " ";
       }
       return 0;
   }

Example - Linear Space

                                         Code
   #include <bits/stdc++.h>
   using namespace std;
   int main() {
       int n;
       cin >> n;
       vector<vector<int>> matrix(n, vector<int>(n));
                                                         The program creates a 2D vector
       for (int i = 0; i < n; i++) {
           for (int j = 0; j < n; j++) {
                                                        (matrix) of size n×n. The space used
                matrix[i][j] = i * n + j;
           }
                                                            grows quadratically with n.
       }
       for (int i = 0; i < n; i++) {
           for (int j = 0; j < n; j++) {
                cout << matrix[i][j] << " ";                         O(n*n)
           }
           cout << endl;
       }
       return 0;
   }

Analyzing Algorithm Efﬁciency

Key Takeaways



     ●   Space Complexity – Refers to how much memory an
         algorithm uses, including input space and extra
         (auxiliary) space.

     ●   Efﬁciency Focus – We compare algorithms based on
         how they scale, not exact values, to choose the most
         optimal one for large inputs.
