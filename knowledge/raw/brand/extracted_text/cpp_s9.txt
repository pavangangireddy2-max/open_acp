     #include<bits/stdc++.h>
     using namespace std;
     class WordFrequencyCounter {
     public:
          void countWords(map<string, int>& wordCount, string& paragraph) {
              stringstream ss(paragraph);
              string word;
              while (ss >> word) {
                  wordCount[word]++;
              }
              for (auto& pair : wordCount) {
                  cout << pair.first << " - " << pair.second << endl;
              }
          }
     };
